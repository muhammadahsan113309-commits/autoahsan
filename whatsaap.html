# Python Flask Server & Chrome Automation
from flask import Flask, request, jsonify
from flask_cors import CORS
import subprocess
import urllib.parse
import time
import pyautogui

app = Flask(__name__)
CORS(app)

CHROME_PATH = r"C:\Program Files\Google\Chrome\Application\chrome.exe"
USER_DATA_DIR = r"C:\Users\%USERNAME%\AppData\Local\Google\Chrome\User Data"

@app.route('/send-whatsapp', methods=['POST'])
def send_whatsapp():
    data = request.json or {}
    phone = data.get('phone')
    message = data.get('message')
    profile = data.get('profile', 'Profile 3')

    if not phone or not message:
        return jsonify({'status': 'error', 'message': 'Missing phone or message'}), 400

    encoded_msg = urllib.parse.quote(message)
    wa_url = f"https://web.whatsapp.com/send?phone={phone}&text={encoded_msg}"

    cmd = [
        CHROME_PATH,
        f"--user-data-dir={USER_DATA_DIR}",
        f"--profile-directory={profile}",
        wa_url
    ]

    subprocess.Popen(cmd)
    time.sleep(7)
    pyautogui.press('enter')

    return jsonify({'status': 'success', 'message': f'Opened WhatsApp with profile {profile}'})

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)

To trigger WhatsApp from your phone outside home network:

  • Website posts message object to Firebase Firestore queue collection.
  • Python script runs firestore.on_snapshot() listener on laptop.
  • When you tap send on mobile web app, Python picks up document from Firestore and triggers Chrome Profile 3!

Complete Laptop Automation Setup

Step 1: Install Python Dependencies

Open Command Prompt (cmd) and run:

pip install flask flask-cors pyautogui

Step 2: Save Python Script

Create folder E:\New folder\automation\ and save the code from the main UI as server.py.

Step 3: Verify Chrome User Profile Path

To ensure Chrome profile directory is named Profile 3:

  1. Open Chrome using Profile 3.
  2. In address bar type chrome://version and press Enter.
  3. Look at Profile Path. It will end with \Profile 3.

Step 4: Automatic Startup on Laptop Boot

Place the batch file into Windows startup so it stays listening in the background.
